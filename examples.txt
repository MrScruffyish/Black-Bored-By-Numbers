# Examples for Black-Bored-By-Numbers.
# Run with: ./board examples.txt examples.mp4 --expand
# The --expand flag is only needed for the algorithm section at the end;
# the arithmetic, variable, square-root and formula examples work without it.

# --- Numbers ---
# A decimal comma or point may be used; spaces group thousands.
1 234,5 + 678,75
5 002,3 - 87,65
4,71 * 13,6
9 175,5 / 25

# --- Simple operators and parentheses ---
# A bare number is a value on its own.
5   # a value, not a calculation
# Each operation is worked out on the same board, left to right.
(1+1)*(2+2.1)
2 + 3 * 4
1/8 * 2

# --- Variables ---
a = 5
a
b = a
b
age = 1+2
x = age + 1
x

# The displayed decimal is what gets stored.
q = 1/3
q * 3

# --- Indexed variables ---
x_123 = 1+4
q_{n+1} = 2       # n is undefined, so this is the literal name q_{n+1}
x_123 * q_{n+1}

# Iterate with an index that has a value.
s_0 = 0
i = 0
s_{i+1} = s_i + i
i = i + 1
s_{i+1} = s_i + i
i = i + 1
s_{i+1} = s_i + i
s_3

# --- Long multiplication with leading zeros skipped ---
0,0012 * 0,0012

# --- Meta commands ---
1 / 3 \meta div_digits=10
x = 1 / 3 \meta div_digits=10

# --- Square roots ---
a = \sqrt{2}
\sqrt{152,29}
\sqrt{2} \meta sqrt_digits=20
\sqrt{152,29} \meta sqrt_digits=6

# --- Math formulas ---
$π \approx 3,141 6$
$x_{n+1} = \frac{1}{2}\left(x_n + \frac{N}{x_n}\right)$

# --- Forced formula block ---
\begin{latex}
x_2 = 2,3333 / 2
\frac{a}{b} = \sqrt{x}
\end{latex}

# --- Value substitution ---
k = 1
$ a_k = 2 $    # writes a_k = 2
$ a_@k = 2 $   # writes a_1 = 2
a_@k = 5
a_@k

# --- Algorithms and the expander (requires --expand) ---
# Ordinary assignments are remembered by the expander.
N = 2
x = 1
q = N / x \meta div_digits=2
s = x + q
x = s / 2
$x_1 = @x$

# A compound expression in one line.
x = (x + N / x) / 2
$x_1 = @x$

# A calculation without a variable still shows its result.
10 + 5

# Repeat with per-iteration division precision. In arithmetic lines the
# loop variable is a bare name; in formula text use @k.
\repeat k = 1..3 digits 2 4 8
q = N / x
x = (x + q) / 2
$x_@k = @x$
\endrepeat

# Conditional block.
\if{N > 1}
$N > 1$
\else
$N \le 1$
\endif

# Algorithm title and global variables.
\begin{algorithm}{Newton-Raphson for @N}
\repeat k = 1..4 digits 2 4 8 16
q = N / x
s = x + q
x = s / 2
$x_@k = @x$
\endrepeat
\end{algorithm}
